local PANEL_BASE_WIDTH = 380

local MIN_SCREEN_WIDTH = 320
local MAX_SCREEN_WIDTH = 1920

local function calculateResponsiveScale(screenSize)
	local screenWidth = screenSize.X
	
	-- Se tela muito pequena, escala proporcionalmente
	if screenWidth < MIN_SCREEN_WIDTH then
		return MIN_SCREEN_WIDTH / PANEL_BASE_WIDTH
	end
	
	-- Se tela muito grande, capped em 1.0
	if screenWidth > MAX_SCREEN_WIDTH then
		return 1.0
	end
	
	-- Escala linear entre min e max
	local normalizedWidth = (screenWidth - MIN_SCREEN_WIDTH) / (MAX_SCREEN_WIDTH - MIN_SCREEN_WIDTH)
	local scale = 0.5 + (normalizedWidth * 0.5) -- Entre 0.5 e 1.0
	
	return math.clamp(scale, 0.5, 1.5)
end

local function createBlur()

end

local function removeBlur()

		blurEffect:Destroy()
		blurEffect = nil
	end
end

local function showNotification(title, iconId, duration)
	local notificationScreenGui = createNotificationGui()
	
	local notificationGui = Instance.new("Frame")
	notificationGui.Name = "NotificationGui"
	notificationGui.Size = UDim2.new(0, 300, 0, 60)
	notificationGui.Position = UDim2.new(1, 20, 1, -100)
	notificationGui.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
	notificationGui.BorderSizePixel = 2
	notificationGui.BorderColor3 = Color3.fromRGB(128, 128, 128)
	notificationGui.Parent = notificationScreenGui

	-- UIScale para notificações responsivas
	local notifScale = Instance.new("UIScale")
	notifScale.Name = "NotificationScale"
	notifScale.Parent = notificationGui

	local notificationIcon = Instance.new("ImageLabel")
	notificationIcon.Name = "NotificationIcon"
	notificationIcon.Size = UDim2.new(0, 40, 0, 40)
	notificationIcon.Position = UDim2.new(0, 10, 0.5, -20)
	notificationIcon.BackgroundTransparency = 1
	notificationIcon.Image = "rbxassetid://" .. iconId
	notificationIcon.Parent = notificationGui

	local notificationLabel = Instance.new("TextLabel")
	notificationLabel.Name = "NotificationLabel"
	notificationLabel.Size = UDim2.new(1, -60, 1, 0)
	notificationLabel.Position = UDim2.new(0, 50, 0, 0)
	notificationLabel.BackgroundTransparency = 1
	notificationLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
	notificationLabel.TextSize = 14
	notificationLabel.Font = Enum.Font.Gotham
	notificationLabel.Text = title
	notificationLabel.TextWrapped = true
	notificationLabel.TextXAlignment = Enum.TextXAlignment.Left
	notificationLabel.Parent = notificationGui

	local progressLine = Instance.new("Frame")
	progressLine.Name = "ProgressLine"
	progressLine.Size = UDim2.new(0, 0, 0, 2)
	progressLine.Position = UDim2.new(0, 0, 1, 0)
	progressLine.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
	progressLine.BorderSizePixel = 0
	progressLine.Parent = notificationGui

	local startTime = tick()
	local totalWidth = 300
	local notificationId = tostring(math.random(100000, 999999))

	notificationTimers[notificationId] = {
		startTime = startTime,
		duration = duration,
		gui = notificationGui,
		screenGui = notificationScreenGui,
		scale = notifScale
	}

	local function updateNotificationScale()
		local screenSize = screenGui.AbsoluteSize
		local scale = calculateResponsiveScale(screenSize)
		notifScale.Scale = scale
	end

	updateNotificationScale()

	screenGui.Changed:Connect(function(property)
		if property == "AbsoluteSize" and notificationTimers[notificationId] then
			updateNotificationScale()
		end
	end)

	local slideInTween = game:GetService("TweenService"):Create(
		notificationGui,
		TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
		{Position = UDim2.new(1, -320, 1, -100)}
	)
	slideInTween:Play()

	local progressConnection
	progressConnection = RunService.RenderStepped:Connect(function()
		if notificationTimers[notificationId] == nil then
			progressConnection:Disconnect()
			return
		end

		local elapsed = tick() - notificationTimers[notificationId].startTime
		local progress = math.min(elapsed / notificationTimers[notificationId].duration, 1)

		if notificationGui and notificationGui.Parent then
			progressLine.Size = UDim2.new(0, totalWidth * progress, 0, 2)
		end

		if progress >= 1 then
			progressConnection:Disconnect()

			if notificationGui and notificationGui.Parent then
				local slideOutTween = game:GetService("TweenService"):Create(
					notificationGui,
					TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
					{Position = UDim2.new(1, 20, 1, -100)}
				)
				slideOutTween:Play()

				slideOutTween.Completed:Connect(function()
					if notificationGui then
						notificationGui:Destroy()
					end
					if notificationScreenGui then
						notificationScreenGui:Destroy()
					end
					notificationTimers[notificationId] = nil
				end)
			end
		end
	end)
end

local dragging = false
local dragStart = nil
local frameStart = nil

panelFrame.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then return end

	if input.UserInputType == Enum.UserInputType.MouseButton1 then
		local relativeY = input.Position.Y - panelFrame.AbsolutePosition.Y

		if relativeY >= 0 and relativeY <= 70 then
			dragging = true
			dragStart = input.Position
			frameStart = panelFrame.Position
		end
	end
end)

UserInputService.InputChanged:Connect(function(input, gameProcessed)
	if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
		local delta = input.Position - dragStart
		panelFrame.Position = UDim2.new(frameStart.X.Scale, frameStart.X.Offset + delta.X, frameStart.Y.Scale, frameStart.Y.Offset + delta.Y)
	end
end)

UserInputService.InputEnded:Connect(function(input, gameProcessed)
	if input.UserInputType == Enum.UserInputType.MouseButton1 then
		dragging = false
	end
end)

local function updateLinksPanelLayout()
	if linksPanel and linksPanelScale then
		local screenSize = screenGui.AbsoluteSize
		local scale = calculateResponsiveScale(screenSize)
		linksPanelScale.Scale = scale
		
		-- Painel principal no centro, links à esquerda com 10px de distância
		local panelCenterX = screenSize.X / 2
		local panelHalfWidth = (PANEL_BASE_WIDTH * scale) / 2
		local linksHalfWidth = (LINKS_PANEL_WIDTH * scale) / 2
		local linksPanelX = panelCenterX - panelHalfWidth - linksHalfWidth - 10
		
		linksPanel.Position = UDim2.new(0, linksPanelX, 0.5, -(LINKS_PANEL_HEIGHT * scale) / 2)
	end
end
